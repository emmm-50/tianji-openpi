cmake_minimum_required(VERSION 3.16)
project(native_ik CXX)
find_package(pinocchio REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(SCENE_FILE "${CMAKE_CURRENT_BINARY_DIR}/marvin_95mm.xml")
set(SHARED_URDF "${CMAKE_CURRENT_SOURCE_DIR}/robot_model/urdf/Pro-m6-696.urdf")
set(SYSTEM_SCENE "/opt/kernelmind/apex/install/marvin_description/share/marvin_description/mjcf/matrix/m6s_696.xml")
add_custom_command(OUTPUT "${SCENE_FILE}"
  COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/build_scene.py" "${SHARED_URDF}" "${SYSTEM_SCENE}" "${SCENE_FILE}"
  DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/build_scene.py" "${SHARED_URDF}" "${SYSTEM_SCENE}"
  VERBATIM)
add_custom_target(native_ik_scene DEPENDS "${SCENE_FILE}")
add_executable(native_ik main.cpp collision_broadphase.cpp)
target_link_options(native_ik PRIVATE "-Wl,--wrap=_ZNK4coal15ComputeDistanceclERKNS_11Transform3sES3_RKNS_15DistanceRequestERNS_14DistanceResultE")
add_dependencies(native_ik native_ik_scene)
target_compile_definitions(native_ik PRIVATE NATIVE_IK_SCENE_FILE="${SCENE_FILE}")
target_compile_features(native_ik PRIVATE cxx_std_17)
target_include_directories(native_ik PRIVATE /opt/kernelmind/apex/install/marvin_qp_controller/include)
target_link_libraries(native_ik PRIVATE /opt/kernelmind/apex/install/marvin_qp_controller/lib/marvin_qp_controller/libqp_controller_core.a pinocchio::pinocchio yaml-cpp -L/home/nvidia/.local/lib/python3.10/site-packages -l:daqp.cpython-310-aarch64-linux-gnu.so /usr/lib/aarch64-linux-gnu/libpython3.10.so)
